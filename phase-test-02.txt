function Show-SfLines([string]$Class, [int]$From, [int]$To) {
  $l = @(Get-Content "force-app\main\default\classes\$Class.cls" -Encoding UTF8)
  "--- $Class $From-$To"
  for ($i = $From; $i -le [Math]::Min($To, $l.Count); $i++) {
    $t = $l[$i - 1].Trim()
    if ($t -and $t -notmatch '^(//|\*|/\*)' -and $t -match 'if\s*\(|for\s*\(|Group|Name|=\s*true|SELECT') {
      if ($t.Length -gt 140) { $t = $t.Substring(0, 140) + '...' }
      "{0,4}  {1}" -f $i, $t
    }
  }
}
